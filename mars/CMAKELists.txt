cmake_minimum_required(VERSION 3.10)
project(MarsGRPC)

# Set C++ standard
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# Find dependencies
find_package(gRPC REQUIRED)
find_package(Protobuf REQUIRED)

# Enable threading (important for gRPC)
find_package(Threads REQUIRED)

# Include necessary directories
include_directories(${CMAKE_CURRENT_BINARY_DIR})
include_directories(${CMAKE_SOURCE_DIR}/proto)

# Set up the gRPC plugin for compiling .proto files
execute_process(COMMAND ${CMAKE_COMMAND} -E make_directory ${CMAKE_BINARY_DIR}/generated)
set(GRPC_CPP_PLUGIN "grpc_cpp_plugin")

# Add .proto file to compile
file(GLOB PROTO_FILES ${CMAKE_SOURCE_DIR}/proto/*.proto)

# Generate C++ source files using the gRPC plugin
foreach(PROTO_FILE ${PROTO_FILES})
    get_filename_component(PROTO_FILE_NAME ${PROTO_FILE} NAME_WE)
    set(PROTO_GEN_FILE ${CMAKE_BINARY_DIR}/generated/${PROTO_FILE_NAME}.pb.cc)
    set(PROTO_GEN_HEADER ${CMAKE_BINARY_DIR}/generated/${PROTO_FILE_NAME}.pb.h)
    set(GRPC_GEN_FILE ${CMAKE_BINARY_DIR}/generated/${PROTO_FILE_NAME}.grpc.pb.cc)
    set(GRPC_GEN_HEADER ${CMAKE_BINARY_DIR}/generated/${PROTO_FILE_NAME}.grpc.pb.h)

    add_custom_command(
        OUTPUT ${PROTO_GEN_FILE} ${PROTO_GEN_HEADER} ${GRPC_GEN_FILE} ${GRPC_GEN_HEADER}
        COMMAND ${Protobuf_PROTOC_EXECUTABLE}
        ARGS --cpp_out=${CMAKE_BINARY_DIR}/generated --grpc_out=${CMAKE_BINARY_DIR}/generated
        --plugin=protoc-gen-grpc=${CMAKE_BINARY_DIR}/grpc_cpp_plugin ${PROTO_FILE}
        DEPENDS ${PROTO_FILE}
    )
    list(APPEND GENERATED_PROTO_SRC ${PROTO_GEN_FILE} ${GRPC_GEN_FILE})
endforeach()

# Add your executable or library
add_executable(MarsGRPC ${GENERATED_PROTO_SRC} src/main.cc)

# Link against gRPC, Protobuf, and Threads
target_link_libraries(MarsGRPC
    gRPC::grpc++
    protobuf::libprotobuf
    Threads::Threads
)
